```python
def welcome():
    print("hello, world")


# 调用函数
welcome()


#         形参↓
def order_test(num, dish):
    print(f"你点了{num}份{dish}")


#     实参↓
order_test(4, "竹笋炒肉")
# 也可以这样子写,xxx=xxx(位置参数需要放在关键字参数的前面)
order_test(num=2, dish="普通的面条")

# /的前面必须得用位置参数,*后面使用关键字参数


def info(name, /, *, age):
    print(name, age)


info("Chen", age=15)

info("Zhan", age=24)


def play(a="参数默认值"):
    print(a)
play(a=1)

# 如果没有传递a的值, 就显示默认值
# 如果有传递a值, 那就用传入的值覆盖默认值
# 默认参数必须要放在必选参数的后面, 或者说: 某个形参, 一旦设置了默认值, 那他后面的所有形参, 也必须要写默认值
```
